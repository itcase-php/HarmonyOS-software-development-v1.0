cmake_minimum_required(VERSION 3.18)
project(hdm_office_lok_probe LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    message(FATAL_ERROR "This probe requires the official OHOS toolchain")
endif()
if(NOT EXISTS "${OFFICE_SOURCE}/include/LibreOfficeKit/LibreOfficeKitInit.h")
    message(FATAL_ERROR "Set OFFICE_SOURCE to the pinned LibreOffice checkout")
endif()
add_executable(hdm_office_lok_probe main.cpp)
target_include_directories(hdm_office_lok_probe SYSTEM PRIVATE "${OFFICE_SOURCE}/include")
target_compile_features(hdm_office_lok_probe PRIVATE cxx_std_17)
target_compile_options(hdm_office_lok_probe PRIVATE -Wall -Wextra -Werror -Wno-error=unused-command-line-argument)
target_link_libraries(hdm_office_lok_probe PRIVATE dl)
