cmake_minimum_required(VERSION 3.24)
project(hdm_ocr_probe LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    message(FATAL_ERROR "Use the HarmonyOS toolchain, not a host/Android compiler")
endif()
if(NOT DEFINED OCR_ENGINE_ROOT)
    message(FATAL_ERROR "Set OCR_ENGINE_ROOT to the matching ABI installation prefix")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${OCR_ENGINE_ROOT}")
set(Tesseract_DIR "${OCR_ENGINE_ROOT}/lib/cmake/tesseract")
set(Leptonica_DIR "${OCR_ENGINE_ROOT}/lib/cmake/leptonica")
find_package(Tesseract 5.5.1 EXACT REQUIRED)
add_library(hdm_ocr_probe SHARED ocr_probe.cpp)
target_compile_features(hdm_ocr_probe PRIVATE cxx_std_17)
target_include_directories(hdm_ocr_probe PRIVATE "${OCR_ENGINE_ROOT}/include")
target_compile_options(hdm_ocr_probe PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-unused-command-line-argument)
target_link_libraries(hdm_ocr_probe PRIVATE Tesseract::libtesseract libace_napi.z.so)
